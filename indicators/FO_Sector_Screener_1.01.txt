// ═══════════════════════════════════════════════════════════════════════════
// F&O SECTOR OI SCREENER 1.01 — for TradingView's Pine Screener (Premium and above)
//   Run it on a watchlist of F&O stocks' CASH symbols (for example NSE:RELIANCE), on an intraday timeframe
//   (15 min recommended: 500 candles = 20 days, enough for relative volume).
//
//   WHAT CHANGED FROM 1.00 (26 Nov 2026)
//     · Sector mapping: every stock is written into the code with its sector (the SECTOR TABLE, one line per stock).
//       Nothing is looked up on TradingView. A stock not in the table reads sector 0.
//     · "Sector filter" dropdown: pick a sector; "In sector" = 1 for its stocks (filter "In sector = 1"), and
//       the preset filter "In chosen sector" does the same.
//     · "Candle" dropdown, as in your other screeners: the latest closed candle, or a time of day in 15-minute steps.
//     · Sector total columns (formula symbols) are OFF by default: they did not load in 1.00. Every stock column
//       works without them.
//
//   THE CANDLE (the "Candle" setting)
//     · by default the LATEST CLOSED candle, or
//     · the time you pick: the candle of the latest day that contains that time (on 15 min, 10:00 picks 10:00–10:15;
//       on 5 min it picks 10:00–10:05). Everything is worked out AS AT THAT CANDLE'S CLOSE. It fills once that
//       candle has closed; until then the columns stay empty (Data = 6). A forming candle is never used.
//
//   SCREENER COLUMNS (the first 10 are shown by default; add the rest with "Manage columns")
//     1  Bias           +2 long build-up · +1 short covering · 0 neutral · −1 long unwinding · −2 short build-up.
//                       Neutral when price or OI moved less than its band (inputs). Sort high to low = bull to bear.
//     2  Price %        cash, the candle's close against the previous day's close
//     3  OI %           near + next month futures OI against the previous day's last prints (roll morning: the one
//                       contract both days share, as in your FCOI Screener)
//     4  OI ₹ cr        that OI change × price, in ₹ crore (assumes TradingView's OI is in shares)
//     5  RVOL %         cash volume from 09:15 to the candle ÷ the average volume from 09:15 to the same time on the
//                       last N earlier days. 100 = normal for this time of day.
//     6  OI % last N    the OI change over the last N candles up to the candle, % of the day's base
//     7  Sector         sector code (list below; the name shows in the chart table and in the Sector filter dropdown)
//     8  In sector      1 = the stock belongs to the sector picked in "Sector filter" (or "All sectors")
//     9  Group key      sort high to low: stocks grouped by sector, and inside each sector ranked by "Rank stocks by".
//                       With sector totals ON, the sectors themselves are ranked too (by "Rank sectors by").
//     10 Data           0 all good · 1 contract-roll morning · 2 the stock's futures OI is missing · 3 stock not in the
//                       sector table · 4 sector totals ON but empty · 6 the picked candle has not closed yet
//     11 vs Nifty %     the stock's price % minus NIFTY 50's
//     12 Nifty %        NIFTY 50 at the candle against its previous close
//     Sector totals (only with "Sector totals" ON):
//     13 Sec OI % · 14 Sec price % · 15 Sec RVOL % · 16 Sec bias · 17 Sec OI ₹ cr · 18 Sec OI % last N ·
//     19 vs Sector % · 20 Sector row (1 for one stock per sector: filter "= 1" for one row per sector)
//
//   SECTOR CODES
//     1 Private Banks · 2 PSU Banks · 3 NBFC and Lending · 4 Capital Markets and Insurance · 5 IT ·
//     6 Pharma and Healthcare · 7 Auto OEM · 8 Auto Components · 9 FMCG · 10 Metals and Mining · 11 Oil and Gas ·
//     12 Power · 13 Capital Goods and Defence · 14 Cement · 15 Realty · 16 Infra and Logistics · 17 Telecom ·
//     18 Consumer Durables · 19 Consumer Services · 20 Chemicals and Paints
//
//   FILTER PRESETS (alert conditions; the screener offers them as yes/no filters)
//     In chosen sector · Long build-up + high RVOL · Short build-up + high RVOL · Short covering + high RVOL ·
//     Quiet OI decline · Against its sector (needs sector totals)
//
//   DATA REQUESTS: 5 at most, the Pine Screener's limit: near-month OI, next-month OI, NIFTY 50, and (only with
//   sector totals ON) the sector OI and sector price formulas. No timeframe, symbol or time inputs. 500 candles.
//   No repainting: every request uses lookahead_off, values come only from closed candles, nothing from later ones.
// ═══════════════════════════════════════════════════════════════════════════

f_bias(float px, float oi, float pb, float ob) =>
    na(px) or na(oi) ? na : math.abs(px) < pb or math.abs(oi) < ob ? 0.0 : (px > 0 ? 1.0 : -1.0) * (oi > 0 ? 2.0 : 1.0)

// Relative volume, cumulative: volume from 09:15 to this candle ÷ the average volume from 09:15 to the same time
// on the last rvDays earlier days. One column per day (a ring), one row per candle slot of the day.
// Works in whichever context it runs: the chart (the stock) or inside a request (the sector formula).
f_rv(float v) =>
    var matrix<float> M = matrix.new<float>(MAXSLOT, rvDays + 1, na)
    var int   col  = 0
    var float cum  = 0.0
    var int   dKey = na
    int   mins = f_min(time)
    float out  = na
    if mins >= 555 and mins < 930
        int k = f_ymd(time)
        if na(dKey) or k != dKey
            dKey := k
            col  := (col + 1) % (rvDays + 1)
            for r = 0 to MAXSLOT - 1
                M.set(r, col, na)
            cum := 0.0
        cum += nz(v)
        int slot = math.min(MAXSLOT - 1, math.floor((mins - 555) / TF_MIN))
        M.set(slot, col, cum)
        float s = 0.0
        int   c = 0
        for j = 0 to rvDays
            if j != col
                float x = M.get(slot, j)
                if not na(x)
                    s += x
                    c += 1
        out := c >= rvMin and s > 0 and cum > 0 ? 100.0 * cum / (s / c) : na
    out

// Inside a requested symbol: this candle, the previous day's last close, today's first open, relative volume,
// and the close nBack candles ago today (na when fewer have passed today)
f_dayReq(int nBack) =>
    var float eod  = na
    var int   eodT = na
    var float dOp  = na
    var array<float> dc = array.new_float()
    if not na(time[1]) and f_ymd(time) != f_ymd(time[1])
        eod  := close[1]
        eodT := time[1]
        dOp  := open
        dc.clear()
    dc.push(close)
    float back = dc.size() > nBack ? dc.get(dc.size() - 1 - nBack) : na
    float rv   = f_rv(volume)
    [time, close, eod, eodT, dOp, rv, back]

// Inside a futures OI series (and Nifty): this candle and the previous day's last close (as FCOI Screener)
f_oiReq() =>
    var float eod  = na
    var int   eodT = na
    if not na(time[1]) and f_ymd(time) != f_ymd(time[1])
        eod  := close[1]
        eodT := time[1]
    [time, close, eod, eodT]

// The stock's own relative volume, every candle (read below only from closed candles)
float rvNow = f_rv(volume)

// ─────────────────────────────── CANDLE AND DAY ───────────────────────────────
// A candle counts only once it has closed. The time test also guards the screener's last candle while it forms.
bool closed = barstate.isconfirmed and (barstate.isrealtime or na(timenow) or time_close <= timenow)
int  bD0    = timestamp(TZ, year(time, TZ), month(time, TZ), dayofmonth(time, TZ), 9, 15, 0)
bool inDay  = time >= bD0 and time < timestamp(TZ, year(time, TZ), month(time, TZ), dayofmonth(time, TZ), 15, 30, 0)
bool cBar   = closed and inDay
var int dayKey = na
bool newDay = cBar and (na(dayKey) or bD0 != dayKey)
// The candle to read: the latest closed one, or the one that contains the picked time
int  slotT  = timestamp(TZ, year(time, TZ), month(time, TZ), dayofmonth(time, TZ), SLOT_H, SLOT_M, 0)
bool isSlot = SLOT_ON and time <= slotT and slotT < time + TF_MS
bool take   = cBar and (not SLOT_ON or isSlot)

// ─────────────────────────────── STATE ───────────────────────────────
var float dClose  = na                     // the day in progress: latest close
var int   dLastT  = na
var float pdClose = na                     // the previous day's last close
var int   pvYmd   = na                     // the previous day's date
// The stock's futures: today's base (the previous day's last prints) and today's latest prints
var bool  stOI  = false
var float sP1   = na
var float sP2   = na
var bool  sRoll = false                    // 1! and 2! switched contracts overnight
var float c1    = na
var float c2    = na
var array<float> dayOI = array.new_float() // today's OI measure at each closed candle
// The sector formulas and Nifty: today's latest closed-candle values
var float kOI   = na
var float kOIE  = na
var int   kOIET = na
var float kOIO  = na
var float kOIB  = na
var float kPx   = na
var float kPxE  = na
var int   kPxET = na
var float kPRv  = na
var float kN    = na
var float kNE   = na
var int   kNET  = na
// The columns, as at the candle being read
var bool  show   = false
var float tPx    = na
var float tOIp   = na
var float tOIcr  = na
var float tOIn   = na
var float tRv    = na
var float tSOIp  = na
var float tSOIcr = na
var float tSOIn  = na
var float tSPx   = na
var float tSRv   = na
var float tNx    = na
var bool  tRoll  = false

// ─────────────────────────────── A NEW DAY ───────────────────────────────
if newDay
    pdClose := dClose
    pvYmd   := na(dLastT) ? na : f_ymd(dLastT)
    dayKey  := bD0
    stOI    := false
    sP1     := na
    sP2     := na
    sRoll   := false
    c1      := na
    c2      := na
    dayOI.clear()
    kOI   := na
    kOIE  := na
    kOIET := na
    kOIO  := na
    kOIB  := na
    kPx   := na
    kPxE  := na
    kPxET := na
    kPRv  := na
    kN    := na
    kNE   := na
    kNET  := na
    if SLOT_ON
        show := false                      // the picked candle of a new day has not closed yet

// ─────────────────────────────── EVERY CLOSED CANDLE OF THE DAY ───────────────────────────────
if cBar
    int today = f_ymd(time)
    // Today's base: the previous day's last OI prints, from the first candle where both futures series have data
    if not stOI and not na(pvYmd) and not na(n1T) and n1T == time and not na(n2T) and n2T == time and not na(n1ET) and not na(n2ET) and f_ymd(n1ET) == pvYmd and f_ymd(n2ET) == pvYmd and n1E > 0 and n2E > 0 and n1C > 0
        stOI  := true
        sP1   := n1E
        sP2   := n2E
        sRoll := math.abs(n1C - n2E) < math.abs(n1C - n1E)   // today's near month looks like yesterday's next month
    // Today's latest futures prints (a futures candle at this exact time; otherwise the last one today stays)
    if not na(n1T) and n1T == time and n1C > 0
        c1 := n1C
    if not na(n2T) and n2T == time and n2C > 0
        c2 := n2C
    // The stock's OI measure: near + next; on a roll morning the one contract both days share
    float base = stOI ? (sRoll ? sP2 : sP1 + sP2) : na
    float m    = stOI ? (sRoll ? c1 : c1 + c2) : na
    if not na(m)
        dayOI.push(m)
    // Sector formulas and Nifty: this candle's values, when their candle starts with this one
    if not na(sT) and sT == time and sC > 0
        kOI   := sC
        kOIE  := sE
        kOIET := sET
        kOIO  := sO
        kOIB  := sB
    if not na(pT) and pT == time and pC > 0
        kPx   := pC
        kPxE  := pE
        kPxET := pET
        kPRv  := pRv
    if not na(xT) and xT == time and xC > 0
        kN    := xC
        kNE   := xE
        kNET  := xET
    // ── The candle being read: every column as at its close ──
    if take
        show  := true
        tPx   := f_pct(close, pdClose)
        tOIp  := f_pct(m, base)
        tOIcr := na(m) or na(base) ? na : (m - base) * close / 1e7
        tOIn  := na(m) or na(base) or base == 0 ? na : 100.0 * (m - (dayOI.size() > lastN ? dayOI.get(dayOI.size() - 1 - lastN) : base)) / base
        tRv   := rvNow
        bool nOk = not na(kN) and not na(kNET) and f_ymd(kNET) < today
        tNx   := nOk ? f_pct(kN, kNE) : na
        // Sector OI formula: base = the previous day's last print; on a roll morning, today's first print
        bool  sOk   = not na(kOI) and not na(kOIET) and f_ymd(kOIET) < today and kOIE > 0
        bool  rollS = sOk and not na(kOIO) and math.abs(kOIO - kOIE) / kOIE > ROLL_TOL
        float sBase = sOk ? (sRoll or rollS ? kOIO : kOIE) : na
        tRoll  := sRoll or rollS
        tSOIp  := f_pct(kOI, sBase)
        tSOIcr := na(kOI) or na(sBase) ? na : (kOI - sBase) / 1e7
        tSOIn  := na(kOI) or na(sBase) or sBase == 0 ? na : 100.0 * (kOI - (na(kOIB) ? sBase : kOIB)) / sBase
        // Sector price formula
        bool pOk = not na(kPx) and not na(kPxET) and f_ymd(kPxET) < today
        tSPx := pOk ? f_pct(kPx, kPxE) : na
        tSRv := pOk ? kPRv : na
    dClose := close
    dLastT := time

// ─────────────────────────────── READINGS AND THE GROUP KEY ───────────────────────────────
float vPx   = show ? tPx : na
float vOIp  = show ? tOIp : na
float vOIcr = show ? tOIcr : na
float vOIn  = show ? tOIn : na
float vRv   = show ? tRv : na
float vNx   = show ? tNx : na
float vSOIp = show and SECQ ? tSOIp : na
float vSOIc = show and SECQ ? tSOIcr : na
float vSOIn = show and SECQ ? tSOIn : na
float vSPx  = show and SECQ ? tSPx : na
float vSRv  = show and SECQ ? tSRv : na
float bias  = f_bias(vPx, vOIp, pxBand, oiBand)
float sBias = f_bias(vSPx, vSOIp, sPxBand, sOiBand)

float secM = keySec == "Sector OI %" ? vSOIp * 100 : keySec == "Sector OI ₹ cr" ? vSOIc : keySec == "Sector price %" ? vSPx * 100 : keySec == "Sector RVOL %" ? vSRv : 0.0
float stkM = keyStk == "OI %" ? vOIp * 10 : keyStk == "OI ₹ cr" ? vOIcr : keyStk == "Price %" ? vPx * 10 : vRv
float major = na(secM) ? 0.0 : math.max(-49999.0, math.min(49999.0, math.round(secM)))
float minor = na(stkM) ? -4999.0 : math.max(-4999.0, math.min(4999.0, math.round(stkM)))
// Major part: the sector's ranking value (0 when sector totals are off), the same for every stock of a sector, so
// they stay together; middle: the sector code; minor: the stock's ranking value.
float gKey = SEC == 0 or not show ? na : major * 1e6 + SEC * 1e4 + (5000 + minor)

int dataCd = not show ? 6 : na(tOIp) ? 2 : SEC == 0 ? 3 : (SECQ and na(tSOIp)) ? 4 : tRoll ? 1 : 0

// ─────────────────────────────── SCREENER COLUMNS ───────────────────────────────
// Not drawn on the chart: the values appear in the Pine Screener (and in a chart's data window).
plot(bias, "Bias", #7E57C2, display=display.all - display.pane - display.price_scale - display.status_line, precision=0)
plot(vPx, "Price %", #2962FF, display=display.all - display.pane - display.price_scale - display.status_line, precision=2)
plot(vOIp, "OI %", #26A69A, display=display.all - display.pane - display.price_scale - display.status_line, precision=2)
plot(vOIcr, "OI ₹ cr", #26A69A, display=display.all - display.pane - display.price_scale - display.status_line, precision=1)
plot(vRv, "RVOL %", #FF9800, display=display.all - display.pane - display.price_scale - display.status_line, precision=0)
plot(vOIn, "OI % last N", #26A69A, display=display.all - display.pane - display.price_scale - display.status_line, precision=2)
plot(SEC, "Sector", #787B86, display=display.all - display.pane - display.price_scale - display.status_line, precision=0)
plot(INSEC ? 1 : 0, "In sector", #787B86, display=display.all - display.pane - display.price_scale - display.status_line, precision=0)
plot(gKey, "Group key", #9E9E9E, display=display.all - display.pane - display.price_scale - display.status_line, precision=0)
plot(dataCd, "Data", #9E9E9E, display=display.all - display.pane - display.price_scale - display.status_line, precision=0)
plot(na(vPx) or na(vNx) ? na : vPx - vNx, "vs Nifty %", #5C6BC0, display=display.all - display.pane - display.price_scale - display.status_line, precision=2)
plot(vNx, "Nifty %", #5C6BC0, display=display.all - display.pane - display.price_scale - display.status_line, precision=2)
plot(vSOIp, "Sec OI %", #00897B, display=display.all - display.pane - display.price_scale - display.status_line, precision=2)
plot(vSPx, "Sec price %", #1E88E5, display=display.all - display.pane - display.price_scale - display.status_line, precision=2)
plot(vSRv, "Sec RVOL %", #FB8C00, display=display.all - display.pane - display.price_scale - display.status_line, precision=0)
plot(sBias, "Sec bias", #7E57C2, display=display.all - display.pane - display.price_scale - display.status_line, precision=0)
plot(vSOIc, "Sec OI ₹ cr", #00897B, display=display.all - display.pane - display.price_scale - display.status_line, precision=0)
plot(vSOIn, "Sec OI % last N", #00897B, display=display.all - display.pane - display.price_scale - display.status_line, precision=2)
plot(na(vPx) or na(vSPx) ? na : vPx - vSPx, "vs Sector %", #1E88E5, display=display.all - display.pane - display.price_scale - display.status_line, precision=2)
plot(SECQ and LEAD ? 1 : 0, "Sector row", #9E9E9E, display=display.all - display.pane - display.price_scale - display.status_line, precision=0)

// ─────────────────────────────── FILTER PRESETS (alert conditions) ───────────────────────────────
bool hiRv = not na(vRv) and vRv >= rvHi
alertcondition(INSEC and SEC > 0, "In chosen sector", "{{ticker}}: in the chosen sector")
alertcondition(bias == 2 and hiRv, "Long build-up + high RVOL", "{{ticker}}: long build-up with high relative volume")
alertcondition(bias == -2 and hiRv, "Short build-up + high RVOL", "{{ticker}}: short build-up with high relative volume")
alertcondition(bias == 1 and hiRv, "Short covering + high RVOL", "{{ticker}}: short covering with high relative volume")
alertcondition(not na(vOIp) and vOIp <= -oiBand and not na(vRv) and vRv < rvHi, "Quiet OI decline", "{{ticker}}: OI falling without high relative volume")
alertcondition(not na(bias) and not na(sBias) and bias * sBias < 0, "Against its sector", "{{ticker}}: reading points the opposite way to its sector")

// ─────────────────────────────── CHART TABLE (checking only) ───────────────────────────────
f_s(float x, int d) => na(x) ? "—" : str.tostring(x, d == 0 ? "#,##0" : d == 1 ? "#,##0.0" : "#,##0.00")
f_bTxt(float b) => na(b) ? "—" : b == 2 ? "+2 long build-up" : b == 1 ? "+1 short covering" : b == -1 ? "−1 long unwinding" : b == -2 ? "−2 short build-up" : "0 neutral"
f_col(float x) => na(x) ? #787B86 : x > 0 ? #26A69A : x < 0 ? #EF5350 : #B2B5BE
var table T = table.new(position.top_right, 2, 22, bgcolor=color.new(#131722, 0), border_color=color.new(#787B86, 60), border_width=1, frame_color=#787B86, frame_width=1)
f_row(int r, string a, string b, color c) =>
    table.cell(T, 0, r, a, text_color=#B2B5BE, text_halign=text.align_left, text_size=size.small)
    table.cell(T, 1, r, b, text_color=c, text_halign=text.align_right, text_size=size.small)
if barstate.islast and tblOn
    string secName = str.split(SEC_NAMES, "|").get(SEC)
    f_row(0,  "FOSS 1.01 · " + syminfo.ticker, str.tostring(SEC) + " " + secName, #D1D4DC)
    f_row(1,  "Candle", SLOT_ON ? slotIn : "latest closed", #D1D4DC)
    f_row(2,  "Bias", f_bTxt(bias), f_col(bias))
    f_row(3,  "Price %", f_s(vPx, 2), f_col(vPx))
    f_row(4,  "OI %", f_s(vOIp, 2), f_col(vOIp))
    f_row(5,  "OI ₹ cr", f_s(vOIcr, 1), f_col(vOIcr))
    f_row(6,  "RVOL %", f_s(vRv, 0), na(vRv) ? #787B86 : vRv >= rvHi ? #FF9800 : #B2B5BE)
    f_row(7,  "OI % last " + str.tostring(lastN), f_s(vOIn, 2), f_col(vOIn))
    f_row(8,  "In sector (" + secIn + ")", INSEC ? "yes" : "no", INSEC ? #26A69A : #787B86)
    f_row(9,  "Group key", f_s(gKey, 0), #B2B5BE)
    f_row(10, "Nifty %", f_s(vNx, 2), f_col(vNx))
    f_row(11, "vs Nifty %", f_s(na(vPx) or na(vNx) ? na : vPx - vNx, 2), f_col(na(vPx) or na(vNx) ? na : vPx - vNx))
    f_row(12, "Sec OI %", SECQ ? f_s(vSOIp, 2) : "off", f_col(vSOIp))
    f_row(13, "Sec price %", SECQ ? f_s(vSPx, 2) : "off", f_col(vSPx))
    f_row(14, "Sec RVOL %", SECQ ? f_s(vSRv, 0) : "off", #B2B5BE)
    f_row(15, "Sec bias", SECQ ? f_bTxt(sBias) : "off", f_col(sBias))
    f_row(16, "Data", str.tostring(dataCd) + (dataCd == 0 ? " all good" : dataCd == 1 ? " roll morning" : dataCd == 2 ? " stock OI missing" : dataCd == 3 ? " not in sector table" : dataCd == 4 ? " sector totals empty" : " candle not closed yet"), dataCd == 0 ? #26A69A : #FF9800)
